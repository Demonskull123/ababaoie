local function getInput()
	local bp = LP:FindFirstChild("Backpack")
	if bp then
		local inp = bp:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end
	local char = LP.Character
	if char then
		local traits = char:FindFirstChild("ServerTraits")
		if traits then
			local inp = traits:FindFirstChild("Input")
			if inp and inp:IsA("RemoteEvent") then return inp end
		end
	end
	local bpTraits = bp and bp:FindFirstChild("ServerTraits")
	if bpTraits then
		local inp = bpTraits:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end
	for _, v in ipairs(LP:GetDescendants()) do
		if v:IsA("RemoteEvent") and v.Name == "Input" then return v end
	end
	return nil
end

local function isSameTeam(model)
	if not farmTeamCheck then return false end
	if not LP.Team then return false end
	local theirPlr = Players:GetPlayerFromCharacter(model)
	if not theirPlr then return false end
	if not theirPlr.Team then return false end
	return theirPlr.Team == LP.Team
end

local function nearestTarget()
	local myChar = LP.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	if not myRoot then return nil end
	local best, bestD = nil, math.huge
	for hum in pairs(hooked) do
		local model = hum.Parent
		if model and model.Parent and model ~= myChar and not isSameTeam(model) then
			local root = model:FindFirstChild("HumanoidRootPart")
			if root then
				local d = (myRoot.Position - root.Position).Magnitude
				if d < bestD then best, bestD = root, d end
			end
		end
	end
	return best
end

local function fireM1()
	local input = getInput()
	if not input then return end
	local char = LP.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local cam = workspace.CurrentCamera
	local target = nearestTarget()
	local hitPos, lookDir
	if target then
		hitPos = target.Position
		lookDir = (target.Position - (root and root.Position or cam.CFrame.Position))
		if lookDir.Magnitude > 0 then lookDir = lookDir.Unit else lookDir = cam.CFrame.LookVector end
	elseif cam then
		local origin = root and root.Position or cam.CFrame.Position
		lookDir = cam.CFrame.LookVector
		hitPos = origin + lookDir * 10
	else
		hitPos = root and (root.Position + root.CFrame.LookVector * 10) or Vector3.new()
		lookDir = root and root.CFrame.LookVector or Vector3.new(0, 0, -1)
	end
	local mouseHit = CFrame.new(hitPos, hitPos + lookDir)
	local down = {[1]="M1",[2]={air=false,md=Vector3.new(0,0,0),mousehit=mouseHit,skeydown=true,skeyreal=false},n=2}
	local up = {[1]="M1",[2]={air=false,md=Vector3.new(0,0,0),mousehit=mouseHit,skeydown=false,skeyreal=true},n=2}
	pcall(function() input:FireServer(unpack(down, 1, down.n or #down)) end)
	task.wait(0.05)
	pcall(function() input:FireServer(unpack(up, 1, up.n or #up)) end)
end

local function fireSkillKey(keyName)
	local kc
	if keyName == "1" then kc = Enum.KeyCode.One
	elseif keyName == "2" then kc = Enum.KeyCode.Two
	elseif keyName == "3" then kc = Enum.KeyCode.Three
	elseif keyName == "4" then kc = Enum.KeyCode.Four
	elseif keyName == "5" then kc = Enum.KeyCode.Five
	else kc = Enum.KeyCode.One end
	pcall(function()
		VirtualInputManager:SendKeyEvent(true, kc, false, game)
		task.wait(0.05)
		VirtualInputManager:SendKeyEvent(false, kc, false, game)
	end)
end

end

function stopHeadRide()
	headRideTarget = nil
	headRideEndTime = 0
	headRidePending = false
	if headRideConn then
		pcall(function() headRideConn:Disconnect() end)
		headRideConn = nil
	end
end

local function startHeadRide(hum)
	if not headRideOn then return end
	if headRideTarget == hum then return end
	headRideTarget = hum
	headRideEndTime = tick() + headRideDuration
	if headRideConn then
		headRideConn:Disconnect()
		headRideConn = nil
	end
	headRideConn = RunService.RenderStepped:Connect(function()
		if not headRideOn then stopHeadRide() return end
		if tick() >= headRideEndTime then stopHeadRide() return end
		local model = hum and hum.Parent
		if not model or not model.Parent then stopHeadRide() return end
		local myChar = LP.Character
		if not myChar then stopHeadRide() return end
		local myRoot = myChar:FindFirstChild("HumanoidRootPart")
		local theirHead = model:FindFirstChild("Head")
		if not myRoot or not theirHead then stopHeadRide() return end
		local vel = theirHead.AssemblyLinearVelocity
		if vel.Y < -60 then stopHeadRide() return end
		local targetCF = theirHead.CFrame * CFrame.new(0, headRideOffset, 0)
		myRoot.CFrame = myRoot.CFrame:Lerp(targetCF, headRideAlpha)
		myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
	end)
end

local function waitForHitThenRide()
	if not headRideOn then return end
	if headRidePending then return end
	headRidePending = true
	local char = LP.Character
	if not char then headRidePending = false return end
	local myRoot = char:FindFirstChild("HumanoidRootPart")
	if not myRoot then headRidePending = false return end
	local best, bestD = nil, math.huge
	for h in pairs(hooked) do
		local model = h.Parent
		if model and model.Parent and model ~= char then
			local r = model:FindFirstChild("HumanoidRootPart")
			if r then
				local d = (r.Position - myRoot.Position).Magnitude
				if d < bestD then best, bestD = h, d end
			end
		end
	end
	if not best then headRidePending = false return end
	local targetModel = best.Parent
	local startHealth = best.Health
	local startTags = {}
	for _, c in ipairs(targetModel:GetChildren()) do startTags[c.Name] = true end
	local conn
	local startTime = tick()
	conn = RunService.Heartbeat:Connect(function()
		if not headRideOn then conn:Disconnect() headRidePending = false return end
		if tick() - startTime > 0.6 then conn:Disconnect() headRidePending = false return end
		local hum = targetModel:FindFirstChildOfClass("Humanoid")
		if not hum then conn:Disconnect() headRidePending = false return end
		local hit = hum.Health < startHealth
		if not hit then
			for _, c in ipairs(targetModel:GetChildren()) do
				if not startTags[c.Name] then
					local n = c.Name:lower()
					if n:find("hit") or n:find("stun") or n:find("slam") or n:find("ragdoll") or n:find("got") then
						hit = true
						break
					end
				end
			end
		end
		if hit then
			conn:Disconnect()
			headRidePending = false
			startHeadRide(hum)
		end
	end)
end

local function blockPulse(duration)
	local input = getInput()
	if not input then return end
	if blockEndTimer then
		pcall(function() blockEndTimer:Cancel() end)
		blockEndTimer = nil
	end
	pcall(function() input:FireServer("BlockOn") end)
	blockEndTimer = task.delay(duration, function()
		local i = getInput()
		if i then pcall(function() i:FireServer("BlockOff") end) end
		blockEndTimer = nil
	end)
end

local function fireBlock()
	local holdTime = abHold + (pingComp and (pingCompMs / 1000) or 0)
	blockPulse(holdTime)
end

local function doTrade()
	if not tradeOn then return end
	if trading then
		if tick() - tradeStartTime > 2 then trading = false else return end
	end
	if tradedThisBlock then return end
	trading = true
	tradedThisBlock = true
	tradeStartTime = tick()
	task.spawn(function()
		pcall(function()
			local holdTime = abHold + (pingComp and (pingCompMs / 1000) or 0)
			task.wait(holdTime + tradeDelay)
			if tradeOn then fireM1() end
		end)
		task.wait(0.6)
		trading = false
	end)
end

local function onAttack()
	if not ABOn and not tradeOn then return end
	fireBlock()
	tradedThisBlock = false
	if tradeOn then doTrade() end
end

local function farmPositionCFrame(targetRoot)
	local pos = targetRoot.Position
	local look = targetRoot.CFrame.LookVector
	if farmPosMode == "On Top" then return CFrame.new(pos + Vector3.new(0, 4, 0))
	elseif farmPosMode == "Below" then return CFrame.new(pos - Vector3.new(0, 4, 0))
	elseif farmPosMode == "Behind" then return CFrame.new(pos - look * 4, pos)
	elseif farmPosMode == "In Front" then return CFrame.new(pos + look * 4, pos) end
	return CFrame.new(pos)
end

local function farmLoop()
	if farmConn then farmConn:Disconnect() farmConn = nil end
	if farmM1Loop then task.cancel(farmM1Loop) farmM1Loop = nil end
	if farmSkillLoop then task.cancel(farmSkillLoop) farmSkillLoop = nil end
	farmRotationIndex = 1
	farmConn = RunService.Heartbeat:Connect(function()
		if not farmOn then
			if farmConn then farmConn:Disconnect() farmConn = nil end
			return
		end
		local char = LP.Character
		if not char then return end
		local myRoot = char:FindFirstChild("HumanoidRootPart")
		if not myRoot then return end
		local best, bestD = nil, farmRange
		for h in pairs(hooked) do
			local model = h.Parent
			if model and model.Parent and model ~= char and not isSameTeam(model) then
				local r = model:FindFirstChild("HumanoidRootPart")
				local hum = model:FindFirstChildOfClass("Humanoid")
				if r and hum and hum.Health > 0 then
					local d = (r.Position - myRoot.Position).Magnitude
					if d < bestD then best, bestD = h, d end
				end
			end
		end
		if not best then
			farmTarget = nil
			forcedLock = nil
			return
		end
		farmTarget = best
		local model = best.Parent
		local targetRoot = model:FindFirstChild("HumanoidRootPart")
		if not targetRoot then return end
		forcedLock = targetRoot
		myRoot.CFrame = farmPositionCFrame(targetRoot)
		myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
	end)
	farmM1Loop = task.spawn(function()
		while farmOn do
			if farmTarget and farmTarget.Parent then
				fireM1()
				task.wait(farmAttackDelay)
			else
				task.wait(0.05)
			end
		end
	end)
	farmSkillLoop = task.spawn(function()
		while farmOn do
			if farmTarget and farmTarget.Parent then fireNextFarmSkill() end
			task.wait(farmSkillDelay)
		end
	end)
end

local function hookHumanoid(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if hooked[hum] then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t = 0
			while hum.Parent and not animator and t < 5 do
				task.wait(0.1); t = t + 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookHumanoid(hum) end
		end)
		return
	end
	hooked[hum] = true
	local model = hum.Parent
	local conn = animator.AnimationPlayed:Connect(function(track)
		if not model or not model.Parent then return end
		local id = track.Animation and track.Animation.AnimationId
		if not id then return end
		local myChar = LP.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end
		if (myRoot.Position - theirRoot.Position).Magnitude > abRange then return end
		if comboEndIds[id] then return end
		if not atkIds[id] then return end
		if not ABOn and not tradeOn then return end
		onAttack()
	end)
	local cleanup
	cleanup = hum.AncestryChanged:Connect(function(_, parent)
		if not parent then
			pcall(function() conn:Disconnect() end)
			pcall(function() cleanup:Disconnect() end)
			hooked[hum] = nil
		end
	end)
end

		if o:IsA("Humanoid") then
			local m = o.Parent

			local h = o.Parent

		plr.CharacterAdded:Connect(function(c)
			if not ABOn and not tradeOn and not headRideOn and not farmOn then return end
			if not hookPlayers then return end
			local h = c:WaitForChild("Humanoid", 5)
			if h then hookHumanoid(h) end
		end)
	end)
end

	end
end

local function nanamiTick()
	if not nanamiOn then return end
	if nanamiFired then return end
	local char = LP.Character
	if not char then return end
	if char:GetAttribute("NANAMIAIM") ~= true then return end
	local cut, goal = findBar()
	if not cut or not goal then return end
	local diff = math.abs(cut.Position.X.Scale - goal.Position.X.Scale)
	if diff <= nanamiTol then
		virtualClick()
		nanamiFired = true
	end
end

local function bindNanamiChar(c)
	if not c then return end
	c:GetAttributeChangedSignal("NANAMIAIM"):Connect(function() nanamiFired = false end)
end

local function yujiTick()
	if not yujiOn then return end
	local cam = workspace.CurrentCamera
	if not cam then return end
	local fov = cam.FieldOfView
	if fov < fovEnter then fovChanged = true end
	if fovChanged and not fovFired and fov >= fovThreshold then
		fireSkillByMode()
		fovFired = true
	end
	if fov > fovReset then
		fovChanged = false
		fovFired = false
	end
end

local function hookYujiKey()
	if yujiKeyConn then
		pcall(function() yujiKeyConn:Disconnect() end)
		yujiKeyConn = nil
	end
	yujiKeyConn = UserInputService.InputBegan:Connect(function(input, gp)
		if gp then return end
		if not yujiOn then return end
		if input.KeyCode ~= yujiKey then return end
		task.spawn(function()
			task.wait(yujiDelay)
			if not yujiOn then return end
			fireSkillByMode()
		end)
	end)
end

local function lockTick()
	local target = forcedLock
	if not target then
		if not lockOn then return end
		local myChar = LP.Character
		if not myChar then return end
		local myRoot = myChar:FindFirstChild("HumanoidRootPart")
		if not myRoot then return end
		local best, bestD = nil, lockRange
		for h in pairs(hooked) do
			local model = h.Parent
			if model and model.Parent and model ~= myChar and not isSameTeam(model) then
				local r = model:FindFirstChild("HumanoidRootPart")
				if r then
					local d = (r.Position - myRoot.Position).Magnitude
					if d < bestD then best, bestD = r, d end
				end
			end
		end
		target = best
	end
	if not target then return end
	local myChar = LP.Character
	if not myChar then return end
	local myRoot = myChar:FindFirstChild("HumanoidRootPart")
	local hum = myChar:FindFirstChildOfClass("Humanoid")
	if not myRoot or not hum then return end
	local p = myRoot.Position
	local t = target.Position
	if hum.AutoRotate then hum.AutoRotate = false end
	myRoot.CFrame = CFrame.lookAt(p, Vector3.new(t.X, p.Y, t.Z))
end

local function doServerHop()
	task.spawn(function()
		local success = pcall(function()
			local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
			local res = game:HttpGet(url)
			local data = HttpService:JSONDecode(res)
			if data and data.data then
				for _, srv in ipairs(data.data) do
					if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers and srv.id ~= game.JobId then
						TeleportService:TeleportToPlaceInstance(game.PlaceId, srv.id, LP)
						return
					end
				end
			end
			TeleportService:Teleport(game.PlaceId, LP)
		end)
		if not success then pcall(function() TeleportService:Teleport(game.PlaceId, LP) end) end
	end)
end

local function doJoinLowest()
	task.spawn(function()
		local success = pcall(function()
			local cursor = ""
			local bestId = nil
			local bestCount = math.huge
			for _ = 1, 3 do
				local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
				if cursor ~= "" then url = url .. "&cursor=" .. cursor end
				local res = game:HttpGet(url)
				local data = HttpService:JSONDecode(res)
				if data and data.data then
					for _, srv in ipairs(data.data) do
						if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers and srv.id ~= game.JobId then
							if srv.playing < bestCount then
								bestCount = srv.playing
								bestId = srv.id
							end
						end
					end
					cursor = data.nextPageCursor or ""
					if cursor == "" then break end
				else
					break
				end
			end
			if bestId then
				TeleportService:TeleportToPlaceInstance(game.PlaceId, bestId, LP)
			else
				TeleportService:Teleport(game.PlaceId, LP)
			end
		end)
		if not success then pcall(function() TeleportService:Teleport(game.PlaceId, LP) end) end
	end)
end

-- LOGGER HOOKS
-- =============================================================
local function hookLog(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if logHooked[hum] then return end
	logHooked[hum] = true
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t = 0
			while hum.Parent and not animator and t < 5 do
				task.wait(0.1); t = t + 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookLog(hum) end
		end)
		return
	end
	local model = hum.Parent
	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end
		if model ~= LP.Character then
			local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
			local theirRoot = model:FindFirstChild("HumanoidRootPart")
			if myRoot and theirRoot then
				local d = (myRoot.Position - theirRoot.Position).Magnitude
				if d <= logRange and not skipNames[anim.Name] then
					task.spawn(function()
						addLogRow(id, anim.Name, model.Name, math.floor(d))
					end)
				end
			end
		end
	end)
end

local function hookMyLogger(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if myLogHooked[hum] then return end
	myLogHooked[hum] = true
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t = 0
			while hum.Parent and not animator and t < 5 do
				task.wait(0.1); t = t + 0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then hookMyLogger(hum) end
		end)
		return
	end
	animator.AnimationPlayed:Connect(function(track)
		local anim = track.Animation
		if not anim then return end
		local id = anim.AnimationId
		if not id or id == "" then return end
		if getgenv().OutlawIncludeLoco then
			task.spawn(function() addMyLogRow(id, anim.Name) end)
		elseif getgenv().OutlawFilterLoco ~= false then
			if not isLocomotion(id, anim.Name) then
				task.spawn(function() addMyLogRow(id, anim.Name) end)
			end
		else
			task.spawn(function() addMyLogRow(id, anim.Name) end)
		end
	end)
end

local function bindMyLoggerChar(char)
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then hookMyLogger(hum) end
	char.ChildAdded:Connect(function(c)
		if c:IsA("Humanoid") then hookMyLogger(c) end
	end)
	char.DescendantAdded:Connect(function(c)
		if c:IsA("Humanoid") then hookMyLogger(c) end
	end)
end

local function hookContainer(container)
	if not container then return end
	for _, c in ipairs(container:GetChildren()) do
		if c:IsA("Model") then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h then hookLog(h) end
		end
	end
end

	local function buildRig()
		if not previewWorld or not previewWorld.Parent then return end
		for _, c in ipairs(previewWorld:GetChildren()) do
			if c.Name ~= "PreviewFloor" then c:Destroy() end
		end
		previewRig, previewHum = nil, nil
		local char = LP.Character
		if not char then return end
		local ok2, clone = pcall(function() return char:Clone() end)
		if not ok2 or not clone then return end
		clone.Name = "PreviewRig"
		for _, d in ipairs(clone:GetDescendants()) do
			if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("Sound") or d:IsA("ParticleEmitter") then
				d:Destroy()
			end
		end
		clone.Parent = previewWorld
		local hrp = clone:FindFirstChild("HumanoidRootPart") or clone.PrimaryPart
		if hrp then
			hrp.Anchored = false
			clone:PivotTo(CFrame.new(0, 3, 0))
			hrp.Anchored = true
		end
		if not hrp then
			local torso = clone:FindFirstChild("Torso")
			if torso then
				clone:PivotTo(CFrame.new(0, 3, 0))
				for _, p in ipairs(clone:GetDescendants()) do
					if p:IsA("BasePart") then p.Anchored = true end
				end
			end
		end
		local hum = clone:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
			local old = hum:FindFirstChildOfClass("Animator")
			if old then old:Destroy() end
			local a = Instance.new("Animator")
			a.Parent = hum
			previewHum = hum
		end
		previewRig = clone
	end

	buildRig()

	local pausedTime = 0

	local function stopPlayback()
		if currentTrack then
			pcall(function() currentTrack:Stop() end)
			currentTrack = nil
		end

	end

	local function playAnim(id)
		if not previewHum or not previewHum.Parent then buildRig() end
		if not previewHum then return end
		stopPlayback()
		local anim = Instance.new("Animation")
		anim.AnimationId = id
		local okA, track = pcall(function() return previewHum:LoadAnimation(anim) end)
		if not okA or not track then return end
		track.Looped = loopPlayback
		track:Play()
		pcall(function() track:AdjustSpeed(playbackSpeed) end)
		currentTrack = track
		currentDuration = track.Length or 0
		paused = false
		btnPause.Text = "Pause"
	end

	local function togglePause()
		if not currentTrack then return end
		paused = not paused
		if paused then
			pausedTime = currentTrack.TimePosition
			pcall(function() currentTrack:AdjustSpeed(0) end)
			pcall(function() currentTrack:Pause() end)
			btnPause.Text = "Resume"
		else
			pcall(function() currentTrack:Play() end)
			pcall(function() currentTrack.TimePosition = pausedTime end)
			pcall(function() currentTrack:AdjustSpeed(playbackSpeed) end)
			btnPause.Text = "Pause"
		end
	end

	RunService.RenderStepped:Connect(function()
		if not currentTrack then return end
		if paused then
			pcall(function() currentTrack.TimePosition = pausedTime end)
		end
		local t = currentTrack.TimePosition or 0
		local dur = currentTrack.Length or currentDuration or 0
		currentDuration = dur
		local frac = dur > 0 and (t / dur) or 0
		if frac > 1 then frac = 1 end
		scrubFill.Size = UDim2.new(frac, 0, 1, 0)
		timeLbl.Text = string.format("%.3fs / %.3fs", t, dur)
	end)

	local scrubDragging = false
	scrubBg.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			scrubDragging = true
		end
	end)
	scrubBg.InputChanged:Connect(function(input)
		if scrubDragging and currentTrack and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local left = scrubBg.AbsolutePosition.X
			local width = scrubBg.AbsoluteSize.X
			if width > 0 then
				local f = math.clamp((input.Position.X - left) / width, 0, 1)
				pausedTime = f * (currentTrack.Length or 0)
				pcall(function() currentTrack.TimePosition = pausedTime end)
			end
		end
	end)
	scrubBg.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			scrubDragging = false
		end
	end)

	end)

scanAll()
local bootLive = workspace:FindFirstChild("Live")
if bootLive then
	bootLive.ChildAdded:Connect(function(c)
		if c:IsA("Model") then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h then hookHumanoid(h) end
		end
	end)
end
workspace.DescendantAdded:Connect(function(o)
	if o:IsA("Humanoid") then
		local m = o.Parent
		local mc = LP.Character
		if not (m and mc and m == mc) then hookHumanoid(o) end
	elseif o:IsA("Animator") then
		local h = o.Parent
		if h and h:IsA("Humanoid") then hookHumanoid(h) end
	end
end)
Players.PlayerAdded:Connect(function(plr)
	plr.CharacterAdded:Connect(function(c)
		local h = c:WaitForChild("Humanoid", 5)
		if h then hookHumanoid(h) end
	end)
end)
LP.CharacterAdded:Connect(function()
	task.wait(1)
	if headRideOn then hookMyAnimator() end
end)
