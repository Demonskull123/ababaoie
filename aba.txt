local function getInput()
	local bp = LP:FindFirstChild("Backpack")
	if bp then
		local inp = bp:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end
	local char = LP.Character
	if char then
		local traits = char:FindFirstChild("ServerTraits")
		if traits then
			local inp = traits:FindFirstChild("Input")
			if inp and inp:IsA("RemoteEvent") then return inp end
		end
	end
	local bpTraits = bp and bp:FindFirstChild("ServerTraits")
	if bpTraits then
		local inp = bpTraits:FindFirstChild("Input")
		if inp and inp:IsA("RemoteEvent") then return inp end
	end
	for _, v in ipairs(LP:GetDescendants()) do
		if v:IsA("RemoteEvent") and v.Name == "Input" then return v end
	end
	return nil
end

local function nearestTarget()
	local myChar = LP.Character
	local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
	if not myRoot then return nil end
	local best, bestD = nil, math.huge
	for hum in pairs(hooked) do
		local model = hum.Parent
		if model and model.Parent and model ~= myChar then
			local root = model:FindFirstChild("HumanoidRootPart")
			if root then
				local d = (myRoot.Position - root.Position).Magnitude
				if d < bestD then best, bestD = root, d end
			end
		end
	end
	return best
end

local function fireM1()
	local input = getInput(); if not input then return end
	local char = LP.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local cam = workspace.CurrentCamera
	local target = nearestTarget()
	local hitPos, lookDir
	if target then
		hitPos = target.Position
		lookDir = (target.Position - (root and root.Position or cam.CFrame.Position))
		if lookDir.Magnitude > 0 then lookDir = lookDir.Unit else lookDir = cam.CFrame.LookVector end
	elseif cam then
		local origin = root and root.Position or cam.CFrame.Position
		lookDir = cam.CFrame.LookVector
		hitPos = origin + lookDir * 10
	else
		hitPos = root and (root.Position + root.CFrame.LookVector * 10) or Vector3.new()
		lookDir = root and root.CFrame.LookVector or Vector3.new(0,0,-1)
	end
	local mouseHit = CFrame.new(hitPos, hitPos + lookDir)
	local d = {[1]="M1",[2]={air=false,md=Vector3.new(0,0,0),mousehit=mouseHit,skeydown=true,skeyreal=false},n=2}
	local u = {[1]="M1",[2]={air=false,md=Vector3.new(0,0,0),mousehit=mouseHit,skeydown=false,skeyreal=true},n=2}
	pcall(function() input:FireServer(unpack(d,1,d.n)) end)
	task.wait(0.05)
	pcall(function() input:FireServer(unpack(u,1,u.n)) end)
end

local function blockPulse(duration)
	local input = getInput(); if not input then return end
	if blockEndTimer then pcall(function() blockEndTimer:Cancel() end) blockEndTimer=nil end
	pcall(function() input:FireServer("BlockOn") end)
	blockEndTimer = task.delay(duration, function()
		local i = getInput()
		if i then pcall(function() i:FireServer("BlockOff") end) end
		blockEndTimer = nil
	end)
end

local function fireBlock()
	local holdTime = abHold + (pingComp and (pingCompMs/1000) or 0)
	blockPulse(holdTime)
end

local function doTrade()
	if not tradeOn then return end
	if trading then
		if tick()-tradeStartTime > 2 then trading=false else return end
	end
	if tradedThisBlock then return end
	trading=true; tradedThisBlock=true; tradeStartTime=tick()
	task.spawn(function()
		pcall(function()
			local holdTime = abHold + (pingComp and (pingCompMs/1000) or 0)
			task.wait(holdTime + tradeDelay)
			if tradeOn then fireM1() end
		end)
		task.wait(0.6); trading=false
	end)
end

local function respondTo(animId)
	local mode = responses[animId] or "Block"
	if mode == "None" then return end
	if mode == "Block" then
		fireBlock()
	elseif mode == "M1Trade" then
		fireBlock(); tradedThisBlock=false
		if tradeOn then doTrade() end
	end
end

local function onAttack(animId)
	if ABOn then fireBlock() end
	tradedThisBlock=false
	if tradeOn then doTrade() end
	if animId then respondTo(animId) end
end

local function hookCombat(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if hooked[hum] then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t=0
			while hum.Parent and not animator and t<5 do task.wait(0.1); t+=0.1 end
			if animator then hookCombat(hum) end
		end); return
	end
	hooked[hum]=true
	local model = hum.Parent
	animator.AnimationPlayed:Connect(function(track)
		if not model or not model.Parent then return end
		local anim = track.Animation; if not anim then return end
		local id = anim.AnimationId
		if not id or not atkIds[id] then return end
		local myChar = LP.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local theirRoot = model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end
		if (myRoot.Position-theirRoot.Position).Magnitude > abRange then return end
		onAttack(id)
	end)
end

local function startCombatHooks()
	local function scan(c)
		if c:IsA("Model") then
			local h = c:FindFirstChildOfClass("Humanoid")
			if h then hookCombat(h) end
		end
	end
	local live = workspace:FindFirstChild("Live")
	if live then for _,c in ipairs(live:GetChildren()) do scan(c) end; live.ChildAdded:Connect(scan) end
	local p = workspace:FindFirstChild("Players")
	if p then for _,c in ipairs(p:GetChildren()) do scan(c) end; p.ChildAdded:Connect(scan) end
	workspace.DescendantAdded:Connect(function(o)
		if o:IsA("Humanoid") then hookCombat(o) end
	end)
end

	-- ===== Rig framing: compute bounding box and frame camera =====
	local function frameRigToViewport(model)
		if not model or not previewCamera or not previewViewport then return end
		-- collect base parts
		local parts = {}
		for _, d in ipairs(model:GetDescendants()) do
			if d:IsA("BasePart") then table.insert(parts, d) end
		end
		if #parts == 0 then return end

		-- bbox
		local minV = Vector3.new(math.huge,math.huge,math.huge)
		local maxV = Vector3.new(-math.huge,-math.huge,-math.huge)
		for _, p in ipairs(parts) do
			local cf = p.CFrame
			local sz = p.Size * 0.5
			-- sample 8 corners in world
			for sx = -1, 1, 2 do
				for sy = -1, 1, 2 do
					for sz2 = -1, 1, 2 do
						local corner = (cf * CFrame.new(sz.X*sx, sz.Y*sy, sz.Z*sz2)).Position
						minV = Vector3.new(math.min(minV.X,corner.X), math.min(minV.Y,corner.Y), math.min(minV.Z,corner.Z))
						maxV = Vector3.new(math.max(maxV.X,corner.X), math.max(maxV.Y,corner.Y), math.max(maxV.Z,corner.Z))
					end
				end
			end
		end
		local center = (minV + maxV) * 0.5
		local size = maxV - minV
		local maxDim = math.max(size.X, size.Y, size.Z)
		-- fov margin
		local fov = previewCamera.FieldOfView
		local aspect = previewViewport.AbsoluteSize.X / math.max(1, previewViewport.AbsoluteSize.Y)
		-- vertical fov in radians
		local vfov = math.rad(fov)
		local hfov = 2 * math.atan(math.tan(vfov/2) * aspect)
		local distV = (size.Y * 0.5) / math.tan(vfov/2)
		local distH = (maxDim * 0.5) / math.tan(hfov/2)
		local dist = math.max(distV, distH) * 1.35 + 2
		-- slight downward + in front
		local lookAt = center
		local camPos = center + Vector3.new(0, size.Y * 0.10, -dist)
		previewCamera.CFrame = CFrame.new(camPos, lookAt)
		previewCamera.FieldOfView = math.clamp(fov, 30, 100)
	end
	getgenv().OutlawFrameRig = frameRigToViewport

	local function buildRig()
		if not previewWorld then return end
		for _, c in ipairs(previewWorld:GetChildren()) do
			if c.Name ~= "PreviewFloor" then c:Destroy() end
		end
		previewRig, previewHum = nil, nil
		local char = LP.Character
		if not char then return end
		local ok2, clone = pcall(function() return char:Clone() end)
		if not ok2 or not clone then return end
		clone.Name = "PreviewRig"
		for _, d in ipairs(clone:GetDescendants()) do
			if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("Sound") or d:IsA("ParticleEmitter") then
				d:Destroy()
			end
		end
		clone.Parent = previewWorld
		-- anchor everything so it stays put and the viewport can frame it
		for _, p in ipairs(clone:GetDescendants()) do
			if p:IsA("BasePart") then
				p.Anchored = true
			end
		end
		local hrp = clone:FindFirstChild("HumanoidRootPart") or clone.PrimaryPart
		local torso = clone:FindFirstChild("Torso") or clone:FindFirstChild("UpperTorso") or clone:FindFirstChild("LowerTorso")
		local pivot = hrp or torso
		if pivot then
			clone:PivotTo(CFrame.new(0, 3, 0))
		end
		local hum = clone:FindFirstChildOfClass("Humanoid")
		if hum then
			hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
			local old = hum:FindFirstChildOfClass("Animator")
			if old then old:Destroy() end
			local a = Instance.new("Animator"); a.Parent = hum
			previewHum = hum
		end
		previewRig = clone
		-- wait a frame for CFrame propagation, then frame
		task.defer(function()
			task.wait(0.05)
			if getgenv().OutlawAutoFit and previewRig then
				frameRigToViewport(previewRig)
			end
		end)
	end
	getgenv().OutlawRebuildRig = buildRig

-- ============================================================
-- SELF HOOK (per-character, robust against respawn)
-- ============================================================
local selfConnByHum = {}

local function attachSelfAnimator(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if selfLogHooked[hum] then return end
	local animator = hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t=0
			while hum.Parent and not animator and t<5 do
				task.wait(0.1); t+=0.1
				animator = hum:FindFirstChildOfClass("Animator")
			end
			if animator then attachSelfAnimator(hum) end
		end)
		return
	end
	selfLogHooked[hum]=true
	local conn = animator.AnimationPlayed:Connect(function(track)
		if not selfLogOn then return end
		local anim=track.Animation; if not anim then return end
		local id=anim.AnimationId
		if not id or id=="" then return end
		if selfBlacklist[id] then return end
		if movementNames[anim.Name] then return end
		local whitelistOnly=getgenv().OutlawSelfWhitelistOnly
		local label
		if whitelistOnly then
			label=selfWhitelist[id]
			if not label then return end
		else
			label=selfWhitelist[id] or (anim.Name~="" and anim.Name or "custom")
		end
		task.spawn(function() addSelfRow(id,label) end)
	end)
	selfConnByHum[hum]=conn
end

function hookSelfLogger()
	local char = LP.Character
	if not char then
		LP.CharacterAdded:Once(function()
			task.wait(0.75)
			hookSelfLogger()
		end)
		return
	end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then
		-- wait for humanoid
		task.spawn(function()
			local t=0
			while char.Parent and not hum and t<5 do
				task.wait(0.1); t+=0.1
				hum = char:FindFirstChildOfClass("Humanoid")
			end
			if hum then attachSelfAnimator(hum) end
		end)
		return
	end
	attachSelfAnimator(hum)
end

function unhookSelfLogger()
	-- we do not disconnect by default so re-enabling is instant, but drop stale hums
	for hum, conn in pairs(selfConnByHum) do
		if not hum.Parent then
			pcall(function() conn:Disconnect() end)
			selfConnByHum[hum]=nil
			selfLogHooked[hum]=nil
		end
	end
end

-- Hard rewire on every respawn: drop old state, reattach after char loads.
LP.CharacterAdded:Connect(function(newChar)
	-- clear per-character hook state so we attach fresh
	for hum, conn in pairs(selfConnByHum) do
		if hum.Parent ~= newChar then
			pcall(function() conn:Disconnect() end)
			selfConnByHum[hum]=nil
			selfLogHooked[hum]=nil
		end
	end
	-- wait for the Animator to exist on the new character
	task.spawn(function()
		local t=0
		local hum
		while newChar.Parent and t<8 do
			task.wait(0.1); t+=0.1
			hum = newChar:FindFirstChildOfClass("Humanoid")
			if hum and hum:FindFirstChildOfClass("Animator") then break end
		end
		if hum then
			-- small settle delay so the default idle has already fired and been filtered
			task.wait(0.4)
			if selfLogOn then attachSelfAnimator(hum) end
		end
	end)
end)

-- ============================================================
-- ENEMY HOOK LOGGER
-- ============================================================
local function hookLog(hum)
	if not hum or not hum:IsA("Humanoid") then return end
	if logHooked[hum] then return end
	logHooked[hum]=true
	local animator=hum:FindFirstChildOfClass("Animator")
	if not animator then
		task.spawn(function()
			local t=0
			while hum.Parent and not animator and t<5 do task.wait(0.1); t+=0.1 end
			if animator then hookLog(hum) end
		end); return
	end
	local model=hum.Parent
	animator.AnimationPlayed:Connect(function(track)
		if not enemyLogOn then return end
		local anim=track.Animation; if not anim then return end
		local id=anim.AnimationId
		if not id or id=="" then return end
		if skipNames[anim.Name] then return end
		if id:find("507766") or id:find("507777") or id:find("507797") then return end
		if model==LP.Character then return end
		local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
		local theirRoot=model:FindFirstChild("HumanoidRootPart")
		if not myRoot or not theirRoot then return end
		local d=(myRoot.Position-theirRoot.Position).Magnitude
		if d>logRange then return end
		task.spawn(function() addLogRow(id,anim.Name,model.Name,math.floor(d)) end)
		if atkIds[id] then
			if (myRoot.Position-theirRoot.Position).Magnitude<=abRange then
				task.spawn(function() respondTo(id) end)
			end
		end
	end)
end

local function hookContainer(container)
	if not container then return end
	for _,c in ipairs(container:GetChildren()) do
		if c:IsA("Model") then
			local h=c:FindFirstChildOfClass("Humanoid")
			if h then hookLog(h) end
		end
	end
end

task.spawn(function()
	hookContainer(workspace:FindFirstChild("Live"))

	local live=workspace:FindFirstChild("Live")
	if live then live.ChildAdded:Connect(function(c)
		if c:IsA("Model") then
			local h=c:FindFirstChildOfClass("Humanoid")
			if h then hookLog(h) end
		end
	end) end
	local p=workspace:FindFirstChild("Players")
	if p then p.ChildAdded:Connect(function(c)
		if c:IsA("Model") then
			local h=c:FindFirstChildOfClass("Humanoid")
			if h then hookLog(h) end
		end
	end) end
	workspace.DescendantAdded:Connect(function(o)
		if o:IsA("Humanoid") then hookLog(o) end
	end)
end)
